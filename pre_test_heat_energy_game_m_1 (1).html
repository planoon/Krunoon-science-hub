ตั้งค่าเชื่อมต่อ Google Sheet (สำหรับครู)

คุณครูสามารถนำคะแนนของนักเรียนไปเก็บบน Google Sheet โดยทำตามขั้นตอนง่ายๆ ดังนี้:

  1. สร้าง Google Sheet ใหม่ 1 แผ่น (แถวแรก: วันเวลา, เลขที่, ชื่อ, ห้อง, คะแนน, เต็ม, ร้อยละ, เวลาที่ใช้, จำนวนครั้งที่สลับหน้าจอ)
  2. ไปที่เมนู ส่วนขยาย (Extensions) > Apps Script
  3. วางโค้ดสคริปต์ด้านล่างนี้ลงไป แล้วกดบันทึก (Save)
  4. กด ทำให้ใช้งานได้ (Deploy) > การทำให้ใช้งานได้ใหม่ (New deployment)
  5. เลือกประเภท: เว็บแอป (Web app), ใครมีสิทธิ์เข้าถึง: ทุกคน (Anyone)
  6. คัดลอก URL ของ Web App ที่ได้ นำมาวางในช่องด้านล่างนี้
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    sheet.appendRow([
      new Date(),
      data.studentId,
      data.name,
      data.room,
      data.score,
      data.total,
      data.percentage,
      data.timeUsed,
      data.tabSwitches
    ]);
    return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}
